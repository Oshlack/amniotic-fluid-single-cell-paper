object <- here("data", "merged", "AmnioSC.Gerli_et_al_2024.Preprocessed_Merged.Seurat.rds")
if (!file.exists(object) | RERUN_ALL) {
  # create Seurat object
  metadata_df <- rbind(seu.AmnioSC@meta.data[,cols_to_merge],
                       seu.Gerli@meta.data[,cols_to_merge])
  
  seu <- CreateSeuratObject(counts = cbind(GetAssayData(object = seu.AmnioSC, assay = "RNA", 
                                                        slot = "counts")[common_genes,],
                                           GetAssayData(object = seu.Gerli, assay = "RNA", 
                                                        slot = "counts")[common_genes,]),
                            meta.data = metadata_df)
  
  seu$early_late_gestation <- ifelse(seu$gest_age_weeks <= 19, "Early",
                                ifelse(seu$gest_age_weeks >= 23, "Late", 
                                       "Mid"))
  
  # rename batch variable
  seu$Dataset_Batch <- seu$Dataset
  seu$Dataset <- NULL
  
  # match our name formatting
  seu$Donor.Paper <- ifelse(seu$Donor %in% names(DONOR.PAPER), DONOR.PAPER[seu$Donor], seu$Donor)
  seu$Sample.Paper <- ifelse(seu$Sample %in% names(SAMPLE.PAPER), SAMPLE.PAPER[seu$Sample], seu$Sample)
  
  # add extra metadata
  combined_metadata <- readr::read_csv(here("data", "Gerli_et_al_2024", "combined_metadata.csv"))
  combined_metadata <- as.data.frame(combined_metadata)
  rownames(combined_metadata) <- combined_metadata$Sample
  
  combined_metadata <- combined_metadata[seu$Donor,]
  rownames(combined_metadata) <- colnames(seu)
  
  stopifnot(combined_metadata$gest_age == seu$gest_age_weeks)
  stopifnot(combined_metadata$foetal_sex == seu$foetal_sex)
  stopifnot(combined_metadata$dataset_batch == seu$Dataset_Batch)
  
  seu@meta.data <- cbind(seu@meta.data, combined_metadata[,c("trimester", "abnormalities", "outcome", "procedure")])
  
  saveRDS(seu, object)
} else {
  seu <- readRDS(object)
}
