object <- here("data", "workflow", "AmnioSC_ALL_optimised_ref.Preprocessed_merged.Seurat.rds")

if (!file.exists(object) | RERUN_ALL) {
  # load in datasets
  sce_data1 <- readRDS(here("data", "workflow", "AmnioSC_01_optimised_ref.Preprocess.SCE.rds"))
  sce_data3 <- readRDS(here("data", "workflow", "AmnioSC_03_optimised_ref.Preprocess.SCE.rds"))
  sce_data4 <- readRDS(here("data", "workflow", "AmnioSC_04_optimised_ref.Preprocess.SCE.rds"))
  sce_data5 <- readRDS(here("data", "workflow", "AmnioSC_05_optimised_ref.Preprocess.SCE.rds"))
  
  # add in batch information
  sce_data1$Dataset <- "dataset_01"
  sce_data3$Dataset <- "dataset_03"
  sce_data4$Dataset <- "dataset_04"
  sce_data5$Dataset <- "dataset_05"

  # distinguish between the cell names of each dataset (prevents duplicate names)
  colnames(sce_data1) <- paste0("01_", colnames(sce_data1))
  colnames(sce_data3) <- paste0("03_", colnames(sce_data3))
  colnames(sce_data4) <- paste0("04_", colnames(sce_data4))
  colnames(sce_data5) <- paste0("05_", colnames(sce_data5))
  
  # identify genes in common between the datasets
  common_genes <- Reduce(intersect, list(rownames(sce_data1),
                                         rownames(sce_data3),
                                         rownames(sce_data4),
                                         rownames(sce_data5)))
  
  sce_data1 <- sce_data1[common_genes,]
  sce_data3 <- sce_data3[common_genes,]
  sce_data4 <- sce_data4[common_genes,]
  sce_data5 <- sce_data5[common_genes,]
  
  sce <- SingleCellExperiment(list(counts = cbind(counts(sce_data1),
                                                  counts(sce_data3),
                                                  counts(sce_data4),
                                                  counts(sce_data5))),
                              colData = data.frame(
                                Sample = c(sce_data1$Sample,
                                           sce_data3$Sample,
                                           sce_data4$Sample,
                                           sce_data5$Sample),
                                Dataset = c(sce_data1$Dataset,
                                            sce_data3$Dataset,
                                            sce_data4$Dataset,
                                            sce_data5$Dataset),
                                Barcode = c(sce_data1$Barcode,
                                            sce_data3$Barcode,
                                            sce_data4$Barcode,
                                            sce_data5$Barcode)))
  
  # create Seurat object
  seu <- CreateSeuratObject(counts = counts(sce),
                                     meta.data = data.frame(Sample = sce$Sample,
                                                            Dataset = sce$Dataset,
                                                            Barcode = sce$Barcode,
                                                            row.names = colnames(sce)))
  
  seu$Donor <- ifelse(seu$Sample %in% c("1AFa", "1AFb"), "1AF", seu$Sample)
  
  saveRDS(seu, object)
} else {
  seu <- readRDS(object)
}

3 Celltypist annotation

Celltypist uses a logistic regression classifier to transfer cell type labels based on an in-built pan-foetal reference dataset that contains cells from a variety of tissues, including foetal skin.

Celltypist was run via command line using the pan foetal reference model and majority voting options.
