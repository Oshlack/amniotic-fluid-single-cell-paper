2 Load in the integrated dataset

seu_int.Gerli <- readRDS(here("data", "Gerli_et_al_2024",
                              "Gerli_et_al_2024.Refined_labels.Seurat.rds"))
seu_int.Gerli$Dataset_Source <- "Gerli et al. (2024)"
seu_int.AmnioSC <- readRDS(here("data", "workflow",
                                "AmnioSC_ALL_optimised_ref.Refined_labels.Seurat.rds"))
seu_int.AmnioSC$Dataset_Source <- "Our Dataset"

seu_int.AmnioSC$Donor <- DONOR.PAPER[seu_int.AmnioSC$Donor]
seu_int.AmnioSC$Donor <- factor(seu_int.AmnioSC$Donor.Paper)
plot_df <- rbind(seu_int.Gerli@meta.data[,c("nCount_RNA", "nFeature_RNA",
                                            "Dataset_Source", "Donor")],
                 seu_int.AmnioSC@meta.data[,c("nCount_RNA", "nFeature_RNA",
                                              "Dataset_Source", "Donor")])
plot_df %>%
  ggplot(aes(x = Donor,
             y = nCount_RNA)) +
  geom_violin(scale = "area") +
  geom_quasirandom(aes(color = Donor)) +
  scale_y_log10() +
  theme_cowplot(12) +
  scale_colour_manual(values = ALL_DONORS_COLOURS.PAPER) +
  NoLegend() +
  rotate_x_labels +
  ggtitle("QC metrics by sample") +
  xlab("Sample") + 
  ylab("Library size") +
  facet_grid(. ~ Dataset_Source, scales = "free_x", space = "free_x") -> p1

plot_df %>%
  ggplot(aes(x = Donor,
             y = nFeature_RNA)) +
  geom_violin(scale = "area") +
  geom_quasirandom(aes(color = Donor)) +
  scale_y_log10() +
  theme_cowplot(12) +
  scale_colour_manual(values = ALL_DONORS_COLOURS.PAPER) +
  NoLegend() +
  rotate_x_labels +
  xlab("Sample") + 
  ylab("Number of detected genes") +
  facet_grid(. ~ Dataset_Source, scales = "free_x", space = "free_x")  -> p2

plot_df %>%
  ggplot(aes(x = Donor,
             fill = Donor)) +
  geom_bar() +
  geom_text(stat="count", aes(label = after_stat(count)), vjust = -0.25) +
  theme_cowplot(12) +
  scale_fill_manual(values = ALL_DONORS_COLOURS.PAPER) +
  NoLegend() +
  ggtitle("Number of cells per sample") +
  rotate_x_labels +
  ylab("Number of cells") +
  xlab("Sample") + 
  facet_grid(. ~ Dataset_Source, scales = "free_x", space = "free_x") -> p3
